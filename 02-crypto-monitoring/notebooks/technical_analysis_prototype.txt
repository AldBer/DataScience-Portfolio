import ccxt
import os
import time
from datetime import datetime

# Configuração segura da exchange
def get_exchange():
    return ccxt.binance({
        'apiKey': os.environ['BINANCE_API_KEY'],
        'secret': os.environ['BINANCE_SECRET_KEY'],
        'enableRateLimit': True,
        'options': {'defaultType': 'spot'}
    })

def check_conditions(exchange, pair='BTC/USDT'):
    try:
        ticker = exchange.fetch_ticker(pair)
        candles = exchange.fetch_ohlcv(pair, '1d', limit=30)
        close_prices = [c[4] for c in candles]
        media_30d = sum(close_prices) / len(close_prices)
        
        return ticker['last'] < media_30d * 0.98
    except Exception as e:
        print(f"Erro na análise: {str(e)}")
        return False

def execute_trade():
    try:
        exchange = get_exchange()
        if check_conditions(exchange):
            order = exchange.create_market_buy_order('BTC/USDT', 0.00001)
            return True, order
        return False, None
    except Exception as e:
        print(f"Erro na execução: {str(e)}")
        raise
